async function analyzeNumber(input) { try { const phoneNumber = libphonenumber.parsePhoneNumber(input, "IN"); if (!phoneNumber) { showInvalid(); return; } // Local basic information document.getElementById("phone").textContent = phoneNumber.number; document.getElementById("country").textContent = phoneNumber.country || "Unknown"; document.getElementById("countryCode").textContent = phoneNumber.country || "Unknown"; document.getElementById("callingCode").textContent = "+" + phoneNumber.countryCallingCode; document.getElementById("numberType").textContent = phoneNumber.getType() || "Unknown"; const valid = phoneNumber.isValid(); document.getElementById("valid").textContent = valid ? "YES" : "NO"; document.getElementById("valid").className = valid ? "ok" : "warning"; document.getElementById("international").textContent = phoneNumber.formatInternational(); document.getElementById("national").textContent = phoneNumber.formatNational(); document.getElementById("possible").textContent = phoneNumber.isPossible() ? "YES" : "NO"; document.getElementById("region").textContent = phoneNumber.country || "Unknown"; // ============================ // NUMVERIFY API // ============================ const response = await fetch( `/api/scan?number=${encodeURIComponent(input)}` ); const data = await response.json(); if (!response.ok) { throw new Error(data.error || "API request failed"); } // Carrier document.getElementById("carrier").textContent = data.carrier || "Not available"; // Timezone let timezone = data.timezone; if (Array.isArray(timezone)) { timezone = timezone.join(", "); } document.getElementById("timezone").textContent = timezone || "Not available"; // Region / location from API if available if (data.location) { document.getElementById("region").textContent = data.location; } // Terminal const terminal = document.getElementById("terminal"); terminal.innerHTML += `
[OK] Number parsed successfully.
[OK] Country: ${escapeHTML( data.country_code || phoneNumber.country || "Unknown" )}
[OK] Calling Code: +${escapeHTML( phoneNumber.countryCallingCode )}
[OK] Valid: ${data.valid ? "YES" : "NO"}
[API] Carrier: ${escapeHTML( data.carrier || "Not available" )}
[API] Location: ${escapeHTML( data.location || "Not available" )}
[INFO] Private identity data is not accessed.
root@mr-error:~$
`; } catch (error) { console.error(error); showInvalid(); document.getElementById("terminal").innerHTML += `
[ERROR] ${escapeHTML(error.message)}
`; } }
